{{- if or .Values.stt.sidecar.enabled (eq .Values.inference.stt.mode "bundled") (eq .Values.inference.stt.mode "external") }}

{{- if or .Values.tts.sidecar.enabled (eq .Values.inference.tts.mode "bundled") (eq .Values.inference.tts.mode "external") }}
